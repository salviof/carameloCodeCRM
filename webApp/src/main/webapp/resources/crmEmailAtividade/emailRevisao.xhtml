/* A etapa fica marcada em azul enquanto o modal dela esta aberto: e o que responde "estou editando esta". A marca e uma classe, e nao :hover nem :focus, de proposito -- o modal abre por cima do ponteiro e o mouseleave nunca chega, entao hover/foco travariam acesos. Quem liga e o clique, quem desliga e o onHide do proprio modal: a marca acompanha o modal, nao o ponteiro. */ function revisaoAbrirEtapa(link, modal) { revisaoLimparEtapas(); var etapa = link.closest('.Atv-exec-email-revisaoEtapa'); if (etapa) { etapa.classList.add('Atv-exec-email-revisaoEtapaEmEdicao'); } /* marca no body: e ela que suspende o hover das outras etapas enquanto o modal cobre a tela. Nao da para usar a mascara do PrimeFaces (.ui-widget-overlay) como sinal: confirmDialog e BlockUI deixam mascaras no DOM o tempo todo, e o hover ficaria suspenso para sempre. */ document.body.classList.add('Atv-exec-email-revisaoModalAberto'); PF(modal).show(); } function revisaoLimparEtapas() { document.body.classList.remove('Atv-exec-email-revisaoModalAberto'); document.querySelectorAll('.Atv-exec-email-revisaoEtapaEmEdicao') .forEach(function (etapa) { etapa.classList.remove('Atv-exec-email-revisaoEtapaEmEdicao'); }); } Sem e-mail
